npm run dev      # start dev server → http://localhost:3000 (Ctrl+C to stop)
npm run build    # production build
npm run start    # serve the production build (run build first)
npm run lint     # eslint, if the scaffold included it

npm install              # install deps (after cloning, or to sync)
npm install <pkg>        # add a dependency
npm install -D <pkg>     # add a dev dependency
